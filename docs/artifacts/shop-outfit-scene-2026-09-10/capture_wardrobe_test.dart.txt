class _FontManifest implements AssetManifest {
  _FontManifest(this.assets);
  final List<String> assets;
  @override
  List<String> listAssets() => assets;
  @override
  List<AssetMetadata>? getAssetVariants(String key) => null;
}

Future<void> _loadFonts() async {
  final aliases = <String, ByteData>{};
  for (final family in ['SpaceGrotesk', 'DMSans']) {
    final data = await rootBundle.load('assets/fonts/$family-Variable.ttf');
    for (final weight in [
      'Thin',
      'ExtraLight',
      'Light',
      'Regular',
      'Medium',
      'SemiBold',
      'Bold',
      'ExtraBold',
      'Black',
    ]) {
      aliases['evidence-fonts/$family-$weight.ttf'] = data;
    }
  }
  final messenger =
      TestDefaultBinaryMessengerBinding.instance.defaultBinaryMessenger;
  messenger.setMockMessageHandler('flutter/assets', (message) async {
    final key = const StringCodec().decodeMessage(message);
    return aliases[key] ??
        await messenger.delegate.send('flutter/assets', message);
  });
  google_base.assetManifest = _FontManifest(aliases.keys.toList());
  GoogleFonts.config.allowRuntimeFetching = false;

  // Register bundled variable fonts under the exact GoogleFonts family aliases
  // used by production TextStyles. This avoids the test binding's Ahem face.
  for (final font in const [
    ('SpaceGrotesk', 'SpaceGrotesk-Variable.ttf'),
    ('DMSans', 'DMSans-Variable.ttf'),
  ]) {
    final data = await rootBundle.load('assets/fonts/${font.$2}');
    for (final suffix in [
      'regular',
      '100',
      '200',
      '300',
      '500',
      '600',
      '700',
      '800',
      '900',
    ]) {
      final loader = FontLoader('${font.$1}_$suffix')
        ..addFont(Future.value(data));
      await loader.load();
    }
    await (FontLoader(font.$1)..addFont(Future.value(data))).load();
  }
  await (FontLoader(
    'Jersey25',
  )..addFont(rootBundle.load('assets/fonts/Jersey25-Regular.ttf'))).load();
  await (FontLoader(
    'MaterialIcons',
  )..addFont(rootBundle.load('fonts/MaterialIcons-Regular.otf'))).load();
}

void main() {
  testWidgets('capture actual wardrobe with fixture accessories', (tester) async {
    PackageInfo.setMockInitialValues(appName:'Bara',packageName:'test',version:'1',buildNumber:'1',buildSignature:'');
    await tester.runAsync(_loadFonts);
    for (final dark in [false, true]) {
      for (final scale in [1.0, 1.6]) {
        tester.view.physicalSize = const Size(390, 844);
        tester.view.devicePixelRatio = 1;
        final preview = PreviewBillingController();
        final api = SectionedAccessoriesApi();
        final controller = CharacterWardrobeController(api: api, auth: preview.auth);
        final boundary = GlobalKey();
        await tester.pumpWidget(RepaintBoundary(key: boundary, child: MaterialApp(
          debugShowCheckedModeBanner: false,
          theme: dark ? AppThemeData.night() : AppThemeData.light(),
          builder: (context, child) => MediaQuery(data: MediaQuery.of(context).copyWith(padding: const EdgeInsets.only(top:44,bottom:34), textScaler: TextScaler.linear(scale)), child: child!),
          home: CharacterWardrobeScreen(character: ShopCharacter.fromJson(api.row('default')), controller: controller, onBuy: (_) async => null),
        )));
        for (var i = 0; i < 5; i++) { await tester.pump(const Duration(milliseconds: 100)); }
        for (final position in ['top', 'selected', 'bottom']) {
          if (position == 'selected') {
            await tester.ensureVisible(find.byKey(const Key('wardrobe-item-accessory-0')));
            await tester.pump();
            await tester.tap(find.byKey(const Key('wardrobe-item-accessory-0')));
            await tester.pump();
            tester.state<ScrollableState>(find.byType(Scrollable).first).position.jumpTo(0);
          }
          if (position == 'bottom') {
            final scroll = tester.state<ScrollableState>(find.byType(Scrollable).first);
            scroll.position.jumpTo(scroll.position.maxScrollExtent);
          }
          for (var i = 0; i < 3; i++) {
            await tester.runAsync(() => Future<void>.delayed(const Duration(milliseconds: 120)));
            await tester.pump(const Duration(milliseconds: 80));
          }
          expect(tester.takeException(), isNull);
          final render = boundary.currentContext!.findRenderObject() as RenderRepaintBoundary;
          await tester.runAsync(() async {
            final image = await render.toImage(pixelRatio: 2);
            final bytes = await image.toByteData(format: ui.ImageByteFormat.png);
            await File('docs/artifacts/shop-outfit-scene-2026-09-10/${dark ? 'dark' : 'light'}-$scale-$position.png').writeAsBytes(bytes!.buffer.asUint8List());
            image.dispose();
          });
        }
        await tester.pumpWidget(const SizedBox.shrink());
        controller.dispose(); preview.dispose();
      }
    }
    tester.view.reset();
  });
}
